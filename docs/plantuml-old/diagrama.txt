'HEADERS COM ESTRUTURAS E FUNÇÕES MEMBRO

struct Tela {
  int largura
  int altura
  const char* titulo

  Tela* Tela_construtor( int a, int l, const char* t )
  void Tela_destrutor( Tela* t )
}

struct CirculoDasHoras {
  float raio
  Vector2 posicao

  CirculoDasHoras* CirculoDasHoras_construtor( Tela* t )
  void CirculoDasHoras_destrutor( CirculoDasHoras* c )
}

struct RaioFixo {
  define NUMERO_RAIOS 23

  Vector2* centro
  Vector2* borda

  RaioFixo* raiosDasHoras_polidestrutor( CirculoDasHoras* c )
  void raiosDasHoras_polidestrutor( RaioFixo* raiosFixos )
}

struct  PonteiroAgora {
  define HORAS_NO_DIA 24
  define MINUTOS_NA_HORA 60

  int8_t gmt
  int hora
  int minuto
  float anguloGraus

  Vector2* centro
  Vector2* bordaMovel

  PonteiroAgora* PonteiroAgora_construtor( CirculoDasHoras* circulo )
  void PonteiroAgora_destrutor( PonteiroAgora* ponteiroAgora )

  void rotacionarHorariamentePonteiroAgora(...)
  void definirGMT(...)
  void carregarHoraMinutoCorrente(...)
  void computarAnguloEmFuncaoDaHora(...)
  void computarAcrescimoDosMinutosParaAnguloDaHoraComputadada(...)
  void renderizarPonteiroDoAgora(...)
}

'HEADERS COM APENAS FUNÇÕES

metaclass componentesangulo {
  define CIRCULO_GRAUS 360

  obterComponenteX( hipotenusa, anguloGraus )
  obterComponenteY( hipotenusa, anguloGraus )
}

metaclass degrad {
  define LOCAL_MAX 360.0
  define LOCAL_MIN 0.0

  float radianoParaGrau( radiano )
  float grauParaRadiano( grau )
  float normalizarPorModuloEmGraus( anguloSelvagem )
}
